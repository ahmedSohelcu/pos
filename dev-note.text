import { createWebHistory, createRouter } from 'vue-router'

const routes = [
    {
        path: '/',
        name: 'home',
        component: () => import('./components/pages/Home.vue')
    },
    {
        path: '/about',
        name: 'about',
        component: () => import('./components/pages/About.vue')
    },
    {
        path: '/:pathMatch(.*)*',
        component: () => import('./components/pages/PageNotFound.vue')
    }
];

const router = createRouter({
    history: createWebHistory(),
    routes,
});

export default router;

<script>
    import { ref } from 'vue';
    export default {
      setup() {
        const isOpen = ref(false);
        
        const toggleSidebar = () => {
          isOpen.value = !isOpen.value;
          
          // Close all other menus when opening a new one
          const treeviews = document.querySelectorAll('.has-treeview');
          treeviews.forEach(treeview => {
            if (!treeview.contains(event.target)) {
              treeview.classList.remove('menu-open');
            }
          });
        };
    
        return {
          isOpen,
          toggleSidebar,
        };
      },
    };
    </script>
