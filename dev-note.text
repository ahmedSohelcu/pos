//------------------------------------------
//03. resources/js/router.js
//------------------------------------------

import { createWebHistory, createRouter } from 'vue-router'

const routes = [
    {
        path: '/',
        name: 'home',
        component: () => import('./components/pages/Home.vue')
    },
    {
        path: '/about',
        name: 'about',
        component: () => import('./components/pages/About.vue')
    },
    {
        path: '/:pathMatch(.*)*',
        component: () => import('./components/pages/PageNotFound.vue')
    }
];

const router = createRouter({
    history: createWebHistory(),
    routes,
});

export default router;


//------------------------------------------
// resources/js/app.js
//------------------------------------------

import { createApp } from "vue";
import router from './router';
import App from "./components/App.vue";

const app = createApp(App);
app.use(router).mount("#app");


//------------------------------------------
// App.vue
//------------------------------------------
<template>
    <div>
        <nav>
            <router-link :to="{ name: 'home' }">Home</router-link>
            <router-link :to="{ name: 'about' }">About</router-link>
        </nav>
        <router-view :key="$route.fullPath"/>
    </div>
</template>

Best Practices

Implement proper loading states for async routes
Add error handling for failed navigation attempts
Consider implementing route guards for protected routes
