<script setup>
import { reactive, computed, onMounted } from 'vue';
import BaseInput from '../../../../ahmed-vue-kit/components/form/BaseInput.vue';
import {
  BRAND_ENDPOINTS,
  CATEGORY_ENDPOINTS,
  UNIT_ENDPOINTS,
  PRODUCT_ENDPOINTS,
  ATTRIBUTE_ENDPOINTS,
} from '../../../../data/endpoint';

import BaseFileUpload from '../../../../ahmed-vue-kit/components/form/BaseFileUpload.vue';

import { useProductStore } from '../store';
import BaseRichTextEditor from '../../../../ahmed-vue-kit/components/form/BaseRichTextEditor.vue';
import api from '../../../../ahmed-vue-kit/api/api';

const productStore = useProductStore();

// fetch data
onMounted(() => {
  // productStore.fetchData();
});

const form = reactive({
  id: null,
  name: '',
  slug: '',
  product_type: 'single',
  tenant_id: 1,

  brand_id: '',
  category_id: '',
  unit_id: '',

  sku: '',
  barcode: '',
  purchase_price: '',
  sale_price: '',
  stock: '',
  track_stock: true,
  alert_quantity: '',
  description: '',

  product_thumbnail: null,
  product_galleries: [],

  is_active: true,
  sorting_order: 0,

  attributes: [{ attribute_id: '', selectable_values: '', values: '' }],
  variants: [],
});

const errors = {};

const isSingle = computed(() => form.product_type === 'single');
const isVariant = computed(() => form.product_type === 'variant');
const isService = computed(() => form.product_type === 'service');

function addAttribute() {
  form.attributes.push({
    attribute_id: '',
    values: '',
  });
}

function removeAttribute(i) {
  form.attributes.splice(i, 1);
}

/**
 * Reset values when attribute changes
 */
function onAttributeChange(index) {
  form.attributes[index].values = '';
  if (form.attributes[index].attribute_id) {
    api
      .get(
        ATTRIBUTE_ENDPOINTS.attributeValues(form.attributes[index].attribute_id)
      )
      .then((res) => {
        form.attributes[index].selectable_values = res.data;
      })
      .catch((err) => {
        console.log(err);
      });
  }
}

function fileSelect() {
  alert('test');
}

function generateVariants() {
  console.log('generate attributes', form.attributes);

  const attrs = form.attributes
    .filter((a) => a.values)
    .map((a) => a.values.split(',').map((v) => v.trim()));

  if (!attrs.length) return;

  const combinations = attrs.reduce((a, b) =>
    a.flatMap((d) => b.map((e) => [].concat(d, e)))
  );

  form.variants = combinations.map((c) => ({
    name: c.join('-'),
    sku: generateSKU(),
    purchase_price: '',
    sale_price: '',
    stock: '',
    variant_thumbnail: null,
  }));
}

function generateSKU() {
  return 'SKU-' + Math.random().toString(36).substring(2, 8).toUpperCase();
}

function removeVariant(i) {
  form.variants.splice(i, 1);
}

function cancel() {
  window.history.back();
}

function selectedFile(file) {
  // console.log('selectedFile', file);
}

const createOrUpate = async () => {
  productStore.errors = {};

  if (productStore.mode === 'edit') {
    await productStore.update(PRODUCT_ENDPOINTS.update(form.id), form);
  } else {
    await productStore.create(PRODUCT_ENDPOINTS.store, form);
  }
};
</script>

              <!-- ATTRIBUTE -->

                  select2
