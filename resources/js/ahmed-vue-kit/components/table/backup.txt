<template>
  <div class="permissions-dashboard p-3">
    <!-- Header -->
    <div class="d-flex justify-content-between align-items-center mb-4">
      <div class="d-flex align-items-center gap-2">
        <i class="fas fa-user-shield text-primary fa-lg"></i>
        <h5 class="mb-0">Role Permissions</h5>
      </div>

      <div class="d-flex align-items-center gap-3">
        <span class="badge bg-primary px-3 py-1">{{ selectedPermissions.length }} Selected</span>
        <div class="form-check form-switch m-0">
          <input
            type="checkbox"
            class="form-check-input"
            :checked="allPermissionsSelected"
            @change="toggleAllPermissions"
          />
          <label class="form-check-label small">Select All</label>
        </div>
      </div>
    </div>

    <!-- Modules Grid -->
    <div class="row g-3">
      <template v-for="module in permissions" :key="module.module">
        <div class="col-lg-4 col-md-6 col-sm-12">
          <div class="module-card shadow-sm">
            <!-- Module Header -->
            <div class="module-header d-flex justify-content-between align-items-center mb-3">
              <div class="d-flex align-items-center gap-2">
                <i class="fas fa-layer-group text-primary"></i>
                <span class="fw-semibold">{{ module.module }}</span>
              </div>
              <div class="form-check form-switch m-0">
                <input
                  type="checkbox"
                  class="form-check-input"
                  :checked="isModuleSelected(module)"
                  @change="toggleModule(module)"
                />
              </div>
            </div>

            <!-- Permissions -->
            <div class="permissions-list">
              <div
                class="permission-item d-flex flex-wrap align-items-center gap-2"
                v-for="action in actions"
                :key="action"
              >
                <span
                  v-if="getPermission(module, action)"
                  :class="'action-badge ' + action"
                  :title="action.toUpperCase()"
                >
                  {{ action }}
                </span>
                <input
                  v-if="getPermission(module, action)"
                  type="checkbox"
                  class="form-check-input ms-auto"
                  :value="getPermission(module, action).id"
                  v-model="selectedPermissions"
                />
              </div>
            </div>
          </div>
        </div>
      </template>
    </div>
  </div>
</template>

<script setup>
import { ref, watch, computed } from 'vue'

const props = defineProps({
  permissions: { type: Array, default: () => [] },
  modelValue: { type: Array, default: () => [] },
})

const emit = defineEmits(['update:modelValue'])

const selectedPermissions = ref([])

const actions = ['view', 'create', 'update', 'delete', 'export', 'import']

// sync with parent
watch(
  () => props.modelValue,
  (val) => {
    selectedPermissions.value = Array.isArray(val) ? [...val] : []
  },
  { immediate: true }
)

watch(
  selectedPermissions,
  (val) => {
    emit('update:modelValue', [...new Set(val)])
  },
  { deep: true }
)

// helpers
function getPermission(module, action) {
  return module.permissions.find((p) => p.name.endsWith(`.${action}`))
}

// MODULE TOGGLE
function getModulePermissionIds(module) {
  return module.permissions.map((p) => p.id)
}

function isModuleSelected(module) {
  const ids = getModulePermissionIds(module)
  return ids.every((id) => selectedPermissions.value.includes(id))
}

function toggleModule(module) {
  const ids = getModulePermissionIds(module)
  const allSelected = ids.every((id) => selectedPermissions.value.includes(id))
  selectedPermissions.value = allSelected
    ? selectedPermissions.value.filter((id) => !ids.includes(id))
    : [...new Set([...selectedPermissions.value, ...ids])]
}

// GLOBAL TOGGLE
function getAllPermissionIds() {
  return props.permissions.flatMap((module) =>
    module.permissions.map((p) => p.id)
  )
}

const allPermissionsSelected = computed(() => {
  const ids = getAllPermissionIds()
  return ids.length && ids.every((id) => selectedPermissions.value.includes(id))
})

function toggleAllPermissions() {
  const ids = getAllPermissionIds()
  selectedPermissions.value = allPermissionsSelected.value ? [] : [...ids]
}
</script>
